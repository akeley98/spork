% Keyword and box macros take optional [only=<spec>, paper=true|false] (think \only<spec>{\sporkRed}{x}):
% without `only`, always formatted; with it, slides format x on <spec> only and the paper formats x
% only if `paper` (default false). Unformatted keywords stay bold and unformatted boxes keep their
% padding, so text doesn't move horizontally between steps (box height and depth still change).
\newif\ifspork@fmtpaper
\pgfkeys{
  /spork/fmt/.is family, /spork/fmt,
  only/.code args={<#1>}{\def\spork@fmtonly{#1}},
  paper/.is if=spork@fmtpaper,
  paper/.default=true,
}
% \spork@fmt{options}{formatted}{unformatted}
\newcommand{\spork@fmt}[3]{%
  \let\spork@fmtonly\relax \spork@fmtpaperfalse
  \pgfqkeys{/spork/fmt}{#1}%
  \ifx\spork@fmtonly\relax
    \expandafter\@firstoftwo
  \else
    \expandafter\spork@fmtalt
  \fi{#2}{#3}}
\ifsporkSlides
  \newcommand{\spork@fmtalt}{\expandafter\alt\expandafter<\spork@fmtonly>}
\else
  \newcommand{\spork@fmtalt}{\ifspork@fmtpaper\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi}
\fi
\newcommand{\spork@key}[3]{\spork@fmt{#1}{\textbf{\color{#2}#3}}{\textbf{#3}}}

% ===========================================================================
% 5. Overlay intent macros
% ===========================================================================
% Every overlay goes through these (the build driver lints bare \only< \uncover< \node< ...).
% The paper is the final, non-animated figure; each element states its paper intent:
%   \sporkReveal<spec>{x}     slides: \uncover<spec> (keeps layout space)   paper: shown
%   \sporkSlideOnly<spec>{x}  slides: \only<spec> (no space when hidden)    paper: removed
%   \sporkPaperOnly{x}        slides: removed                              paper: shown
% <spec> is optional (default: all slides). Usable in text and inside tikzpicture
% (wrapping whole \node/\draw commands; then the element doesn't exist when removed).
% For TikZ elements that should keep their size/anchors when hidden, prefer the keys in section 6.
\ifsporkSlides
  \DeclareRobustCommand{\sporkReveal}{\@ifnextchar<{\spork@reveal}{\spork@reveal<1->}}
  \def\spork@reveal<#1>#2{\uncover<#1>{#2}}
  \DeclareRobustCommand{\sporkSlideOnly}{\@ifnextchar<{\spork@slideonly}{\spork@slideonly<1->}}
  \def\spork@slideonly<#1>#2{\only<#1>{#2}}
  \newcommand{\sporkPaperOnly}[1]{}
\else
  \DeclareRobustCommand{\sporkReveal}{\@ifnextchar<{\spork@reveal}{\spork@reveal<>}}
  \def\spork@reveal<#1>#2{#2}
  \DeclareRobustCommand{\sporkSlideOnly}{\@ifnextchar<{\spork@slideonly}{\spork@slideonly<>}}
  \def\spork@slideonly<#1>#2{}
  \newcommand{\sporkPaperOnly}[1]{#1}
\fi

% Overlay intent keys (TikZ counterpart of section 5):
%   spork reveal=<spec>             slides: invisible outside <spec> (keeps size, anchors, bbox)  paper: shown
%   spork slide only=<spec>         slides: same as reveal                                         paper: invisible, not in bbox
%   spork slide only=<spec>{keys}   slides: apply keys on <spec>                                   paper: ignored
%   spork paper only                slides: invisible, not in bbox                                 paper: shown
%   spork paper style={keys}        slides: ignored                                                paper: apply keys
% "Invisible" = opacity 0: the element still exists (anchors usable by other elements), and its text is
% still in the PDF text layer. To remove an element entirely, wrap the whole command in \sporkSlideOnly / \sporkPaperOnly.
\tikzset{
  spork@hidden/.style={opacity=0, text opacity=0},
  spork@absent/.style={spork@hidden, overlay},
}
\ifsporkSlides
  \tikzset{
    spork reveal/.code args={<#1>}{\alt<#1>{}{\pgfkeysalso{spork@hidden}}},
    spork slide only/.code args={<#1>#2}{%
      \ifx\relax#2\relax
        \alt<#1>{}{\pgfkeysalso{spork@hidden}}%
      \else
        \alt<#1>{\pgfkeysalso{#2}}{}%
      \fi},
    spork paper only/.style={spork@absent},
    spork paper style/.code={},
  }
\else
  \tikzset{
    spork reveal/.code args={<#1>}{},
    spork slide only/.code args={<#1>#2}{%
      \ifx\relax#2\relax\pgfkeysalso{spork@absent}\fi},
    spork paper only/.code={},
    spork paper style/.code={\pgfkeysalso{#1}},
  }
\fi

% ===========================================================================
% 9. sporkfigure environment
% ===========================================================================
% \begin{sporkfigure}[scale=S, <tikzpicture options>] ... \end{sporkfigure}
% A tikzpicture with `spork figure` (fixed sans \sporkFigFont), typeset into a box and then scaled as a
% whole (text included) by S with \scalebox. Default S = \sporkFigScale = 1 in both modes (decided
% 2026-10-25: \sporkFigFont is 11pt, the paper body size, so slides need no compensating scale; use
% `scale` only for real layout reasons such as fitting a column). The body has the same geometry in
% paper and slides; `scale` only magnifies it. Inline like a tikzpicture: use it inside figure/center/columns as you would a picture.
% `scale=` in the options is the sporkfigure scale (not TikZ's coordinate scale); `scale=` on scopes,
% nodes or styles inside the body is TikZ's as usual. Overlay keys inside animate per Beamer step.
%
% Width warning: if the scaled figure is wider than \linewidth (of the place it is used: paper column,
% frame, Beamer column), emits
%   Package spork Warning: sporkfigure is <w> wide after scaling, > \linewidth <lw>
% `remember picture` on the sporkfigure also warns: positions recorded inside a scaled box are wrong
% under xelatex (listing marks and decorations go in plain pictures outside, see README Marks).
%
% Local node names. Stock TikZ keeps every node name for the rest of the document, and a picture
% referring to a name it doesn't define silently uses whatever picture defined it last (no error;
% `name prefix` also silently falls back to the unprefixed name). Inside a sporkfigure:
%   (1) Every name created in it (nodes, coordinates, aliases, pic and matrix nodes, also inside
%       nested pictures) is forgotten (made undefined) at \end{sporkfigure}. A later reference to it,
%       from any picture, is pgf's normal error: No shape named `C2' is known.
%   (2) A reference to a node that was defined in a different picture (a plain tikzpicture earlier in the
%       document, e.g. a listing-mark overlay or an inline \tikz) is also an error:
%       No shape named `C2' is known in this sporkfigure (that name belongs to another picture).
%       Exempt: `current page` (not meaningful in a scaled box anyway) and the per-picture
%       `current bounding box` etc. A nested picture (\tikz inside a node) is "another picture" too.
% So a sporkfigure can only use names it defines itself; two sporkfigures may use the same names freely.
% Names of plain tikzpictures are not forgotten (stock behavior outside sporkfigures, which
% remember-picture overlays rely on), except when a sporkfigure later reuses the same name (rule 1).
%
% PGF internals relied on (pgf 3.1.x, generic/pgf/modules/pgfmoduleshapes.code.tex):
%   * a node named N is the macros \pgf@sh@{ns,np,nt,pi,ma,sa}@N (shape, saved points, transform, picture
%     id, saved macros, saved anchor; TikZ adds \tikz@dcl@coord@N); undefining them forgets N;
%   * \pgf@nodecallback{N} is called for every created node (\pgfmultipartnode, also late-positioned) and
%     alias (\pgfnodealias; aliases, `alias=`, renames); \pgfpicture resets it to \pgfutil@gobble and
%     matrix cells replace it with \pgf@matrix@nodecallback, so both are wrapped; \pgfcoordinate
%     (no callback) is wrapped too;
%   * \pgf@shape@interpictureshift{N} is called by every anchor/border lookup (\pgfpointanchor,
%     \pgfpointshapeborder) and compares \pgf@sh@pi@N with \pgfpictureid: the hook for rule (2).
% All wrappers are local to the sporkfigure (plain pictures run stock pgf code). Not tracked: names made
% by `local bounding box` and by \pgffakenode (still covered by rule 2 if referenced from a later figure).
\newcommand{\sporkFigScale}{1}
\newsavebox{\spork@figbox}
\newif\ifspork@infig
\gdef\spork@fig@names{}
\def\spork@fig@pageorigin{pgfpageorigin}
% Record name #1 (already expanded) for forgetting at \end{sporkfigure}.
\def\spork@fig@track#1{%
  \def\spork@tmp{#1}%
  \ifx\spork@tmp\pgfutil@empty\else
    \xdef\spork@fig@names{\unexpanded\expandafter{\spork@fig@names}\noexpand\spork@fig@forget{\unexpanded{#1}}}%
  \fi}
% \spork@fig@neverdefined is never defined: \let to it makes a macro undefined again (also for \ifcsname,
% which TikZ uses for the `name prefix` fallback; \let to \relax would not be).
\def\spork@fig@undef#1{\expandafter\global\expandafter\let\csname #1\endcsname\spork@fig@neverdefined}
\def\spork@fig@forget#1{%
  \spork@fig@undef{pgf@sh@ns@#1}\spork@fig@undef{pgf@sh@np@#1}\spork@fig@undef{pgf@sh@nt@#1}%
  \spork@fig@undef{pgf@sh@pi@#1}\spork@fig@undef{pgf@sh@ma@#1}\spork@fig@undef{pgf@sh@sa@#1}%
  \spork@fig@undef{tikz@dcl@coord@#1}}
% Rule (2): wraps \pgf@shape@interpictureshift inside sporkfigures.
\def\spork@fig@interpictureshift#1{%
  \edef\spork@tmp{\csname pgf@sh@pi@#1\endcsname}%
  \ifx\spork@tmp\pgfpictureid\else
    \ifx\spork@tmp\spork@fig@pageorigin\else
      \pgferror{No shape named `#1' is known in this sporkfigure (that name belongs to another picture)}%
    \fi
  \fi
  \spork@fig@orig@interpictureshift{#1}}
\tikzset{
  % First and last of the sporkfigure's picture options: `scale=` in between is captured as the figure scale.
  spork@fig@options begin/.code={%
    \pgfkeysgetvalue{/tikz/scale/.@cmd}\spork@fig@saved@scale
    \pgfkeysdef{/tikz/scale}{\xdef\spork@fig@scale{##1}}},
  spork@fig@options end/.code={%
    \pgfkeyslet{/tikz/scale/.@cmd}\spork@fig@saved@scale},
  spork@fig@track nodes/.code={\let\pgf@nodecallback\spork@fig@track},
}
\newenvironment{sporkfigure}[1][]{%
  \ifspork@infig
    \PackageError{spork}{sporkfigure inside a sporkfigure is not supported}{}%
  \fi
  \spork@infigtrue
  \xdef\spork@fig@scale{\sporkFigScale}%
  \gdef\spork@fig@names{}%
  % Wrappers (local to this environment; see above).
  \let\spork@fig@orig@interpictureshift\pgf@shape@interpictureshift
  \let\pgf@shape@interpictureshift\spork@fig@interpictureshift
  \let\spork@fig@orig@coordinate\pgfcoordinate
  \def\pgfcoordinate##1##2{\edef\spork@tmpb{##1}%
    \expandafter\spork@fig@track\expandafter{\spork@tmpb}\spork@fig@orig@coordinate{##1}{##2}}%
  \let\spork@fig@orig@matrixcb\pgf@matrix@nodecallback
  \def\pgf@matrix@nodecallback##1{\spork@fig@track{##1}\spork@fig@orig@matrixcb{##1}}%
  % Every picture inside (the figure and nested ones): \pgfpicture just reset the callback to gobble.
  \tikzset{every picture/.append style={spork@fig@track nodes}}%
  \begin{lrbox}{\spork@figbox}%
  \begin{tikzpicture}[spork@fig@options begin, spork figure, #1, spork@fig@options end,
      execute at end picture={\ifpgfrememberpicturepositiononpage
        \PackageWarning{spork}{remember picture in a sporkfigure: recorded positions are wrong
          under scaling; draw marks/decorations in a plain picture outside the sporkfigure}\fi}]%
}{%
  \end{tikzpicture}%
  \end{lrbox}%
  \spork@fig@names
  \gdef\spork@fig@names{}%
  \sbox\spork@figbox{\scalebox{\spork@fig@scale}{\usebox\spork@figbox}}%
  \ifdim\wd\spork@figbox>\linewidth
    \PackageWarning{spork}{sporkfigure is \the\wd\spork@figbox\space wide after scaling,
      > \string\linewidth\space\the\linewidth}%
  \fi
  \usebox\spork@figbox
}
\makeatother
