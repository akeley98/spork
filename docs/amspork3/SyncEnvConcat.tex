\magicSubsection{Synchronization Environment Concatenation}{sec:SyncEnvConcat}
% ``concatenation'' maybe dumb

Define helper
\begin{align}
    & \sporksyncsWith: \sporkEnvVis \times \sporkTr \to \sporkBool
    \label{eq:syncsWith}
    \\
    & \sporksyncsWith((\sporkOb_v, \sporkAw_v, \sporkQtlMask), \sporktr) =
    \sporkOb_v \cap \sporkObIn(\sporktr) \cap (\sporkTid \times \sporkQtlMask) \ne \varnothing
      \lor
      \sporkAw_v \cap \sporkAwIn(\sporktr) \ne \varnothing \notag % f the overfull hbox
\end{align}
Define $\sporkenv \sporkConcatEnv \sporktr_d$ by cases; all cases are total except $\sporkTrReqCanAccess$:

\mainKey{Case 1/3:} $\sporktr_d = \sporkTrNewAccess(\sporkmut, \sporkx, \sporkz_i^*, \sporkTidNew, \sporkqtlInit, \sporkQtlAtom, \sporkQtlMask, \sporkAwNew)$
\begin{align}
    \sporkenv \sporkConcatEnv \sporktr_d =
    % \sporkenv \cup \{ ((\sporkmut, \sporkx, \sporkz_i^*), (\sporkTidNew \times \{\sporkqtlInit\}, \sporkAwNew, \sporkQtlMask)) \}
    \sporkenv \cup \{ ((\sporkmut, \sporkx, \sporkz_i^*), (\sporkObOut(\sporktr_d) \cap (\sporkTid \times \sporkQtlMask), \sporkAwOut(\sporktr_d), \sporkQtlMask)) \}
    \label{eq:envConcatNew}
\end{align}
\mainKey{Case 2/3:} $\sporktr_d$ is constructed with $\sporkTrFence$, $\sporkTrArrive$, or $\sporkTrAwait$.
\begin{align}
    & \sporkenv \sporkConcatEnv \sporktr_d = \{ (\sporkenvEff, \lambda(\sporkenvVis)) \mid (\sporkenvEff, \sporkenvVis) \in \sporkenv \}
    \label{eq:envConcatSync} \\
    % Idiotic unpacking of \sporkenvVis for now.
    % PL community should invest in standardized math syntax that's actually good.
    & \text{where } \lambda(\sporkenvVis) = \lambda((\sporkOb_v, \sporkAw_v, \sporkQtlMask)) = \notag \\
    & \quad \begin{cases}
      (\sporkOb_v \cup (\sporkObOut(\sporktr_d) \cap (\sporkTid \times \sporkQtlMask)), \sporkAw_v \cup \sporkAwOut(\sporktr_d), \sporkQtlMask)
      & \sporksyncsWith(\sporkenvVis, \sporktr_d) \\
      \sporkenvVis & \text{otherwise}
    \end{cases} \notag
\end{align}
\mainKey{Case 3/3:} $\sporktr_d = \sporkTrReqCanAccess(\sporkmut, \sporkx, \sporkz_i^*, \sporkTidReq, \sporkQtlPre, \sporkAwFree)$
\begin{align}
    & \sporkenv \sporkConcatEnv \sporktr_d = \begin{cases}
        \sporkenv & \textsf{isLegal} \\
        \emph{undefined} & \text{otherwise}
    \end{cases}
    \label{eq:envConcatReq}
    \\
    \text{where \textsf{isLegal} = }
    & \textbf{let } \sporkEnvEff_e = \sporkmutReqs(\sporkmut) \times \{ \sporkx \} \times \{ \sporkz_i^* \} \textbf{ in} \notag \\
    & \forall \sporkenvEff \in \sporkEnvEff_e. \notag \\
    & \forall \sporkenvVis \in \sporkenv(\sporkenvEff).\,
    \sporksyncsWith(\sporkenvVis, \sporktr_d) \notag
\end{align}
