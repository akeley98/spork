\sporkSubsection{Synchronization Environment Concatenation}{sec:SyncEnvConcat}
% ``concatenation'' maybe dumb

Define helpers
\begin{align}
    & \sporksyncsWith: \sporkEnvVis \times \sporkTr \to \sporkBool
    \label{eq:syncsWith}
    \\
    & \sporksyncsWith((\sporkOb_v, \sporkAw_v, \sporkQtlMask), \sporktr) =
    \sporkOb_v \cap \sporkObIn(\sporktr) \cap (\sporkTid \times \sporkQtlMask) \ne \varnothing
      \lor
      \sporkAw_v \cap \sporkAwIn(\sporktr) \ne \varnothing \notag % f the overfull hbox
    \\
    & \sporkConcatVis: \sporkEnvVis \times \mathcal{P}(\sporkTr) \to \sporkEnvVis
    \label{eq:ConcatVis}
    \\
    & (\sporkOb_v, \sporkAw_v, \sporkQtlMask) \sporkConcatVis \sporkTr_a =
    (\sporkOb_v \cup (\sporkObOut(\sporkTr_a) \cap (\sporkTid \times \sporkQtlMask)), \sporkAw_v \cup \sporkAwOut(\sporkTr_a), \sporkQtlMask)
\end{align}

Define $\sporkenv \sporkConcatEnv \sporktr_d$ by cases; all cases are total except $\sporkTrPreAccess$:

\sporkMainKey{Case 1/3:} $\sporktr_d = \sporkTrPostAccess(\sporkmut, \sporkx, \sporkz_i^*, \sporkTidPost, \sporkqtlInit, \sporkQtlAtom, \sporkQtlMask, \sporkAwPost)$
\begin{align}
    \sporkenv \sporkConcatEnv \sporktr_d =
    % \sporkenv \cup \{ ((\sporkmut, \sporkx, \sporkz_i^*), (\sporkTidPost \times \{\sporkqtlInit\}, \sporkAwPost, \sporkQtlMask)) \}
    \sporkenv \cup \{ ((\sporkmut, \sporkx, \sporkz_i^*), (\varnothing, \varnothing, \sporkQtlMask) \sporkConcatVis \{ \sporktr_d \}) \}
    \label{eq:envConcatPost}
\end{align}
\sporkMainKey{Case 2/3:} $\sporktr_d$ is constructed with $\sporkTrFence$, $\sporkTrArrive$, or $\sporkTrAwait$.
\begin{align}
    & \sporkenv \sporkConcatEnv \sporktr_d = \{ (\sporkenvEff, \lambda(\sporkenvVis)) \mid (\sporkenvEff, \sporkenvVis) \in \sporkenv \}
    \label{eq:envConcatSync} \\
    & \text{where } \lambda(\sporkenvVis) = \begin{cases}
      \sporkenvVis \sporkConcatVis \{ \sporktr_d \}
      & \sporksyncsWith(\sporkenvVis, \sporktr_d) \\
      \sporkenvVis & \text{otherwise}
    \end{cases} \notag
\end{align}
\sporkMainKey{Case 3/3:} $\sporktr_d = \sporkTrPreAccess(\sporkmut, \sporkx, \sporkz_i^*, \sporkTidPre, \sporkQtlPre, \sporkAwFree)$
\begin{align}
    & \sporkenv \sporkConcatEnv \sporktr_d = \begin{cases}
        \sporkenv & \sporkisLegalEnv \\
        \emph{undefined} & \text{otherwise}
    \end{cases}
    \label{eq:envConcatPre}
    \\
    \text{where $\sporkisLegalEnv$ = }
    & \textbf{let } \sporkEnvEff_e = \sporkmutReqs(\sporkmut) \times \{ \sporkx \} \times \{ \sporkz_i^* \} \textbf{ in} \notag \\
    & \forall \sporkenvEff \in \sporkEnvEff_e. \notag \\
    & \forall \sporkenvVis \in \sporkenv(\sporkenvEff).\,
    \sporksyncsWith(\sporkenvVis, \sporktr_d) \notag
\end{align}
The $\sporkTrPostAccess$ behavior acts as a ``postcondition'' in that in controls how $\sporkenv$ is modified,
and the $\sporkTrPreAccess$ behavior acts as a ``precondition'' that must be satisfied to allow progress.
