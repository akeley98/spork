% \ropeBar{N}{S1}{S2}{S3}{S4}: the four "ropes" (bug classes) bar, first used in "Bugs on the Horizon".
% Each argument is an overlay spec (use 0 for "never"):
%   N  = steps where the bar is neutral (all black text, light gray bars);
%   Sk = steps where rope k is on (purple text, thick purple bar); wins over N.
% On any other step, rope k is off (gray text, light gray bar).
% That's the 17 states: neutral, or any on/off combination of the 4 ropes.
\newcommand{\ropeTab}[4]{% #1 = neutral steps, #2 = on steps, #3 = heading, #4 = name
  {\color{sporkGrayttColor}\sporkSlideOnly<#1>{\usebeamercolor[fg]{normal text}}%
  \sporkSlideOnly<#2>{\color{sporkMainColor}}%
  \textbf{#3}\\#4\par}%
  \vspace{0.8mm}%
  {\color{sporkGrayttColor!35}\sporkSlideOnly<#2>{\color{sporkMainColor}}\rule{\linewidth}{1.2mm}}%
}
\newcommand{\ropeBar}[5]{%
  \begin{columns}[T]
  \begin{column}{0.24\textwidth}\ropeTab{#1}{#2}{Rope 1/4:}{Dataflow Logic}\end{column}
  \begin{column}{0.24\textwidth}\ropeTab{#1}{#3}{Rope 2/4:}{Instruction Usage}\end{column}
  \begin{column}{0.24\textwidth}\ropeTab{#1}{#4}{Rope 3/4:}{Inter-thread Sync}\end{column}
  \begin{column}{0.24\textwidth}\ropeTab{#1}{#5}{Rope 4/4:}{Intra-thread Sync}\end{column}
  \end{columns}%
}
