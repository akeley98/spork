% \riskBar{N}{S1}{S2}{S3}{S4}: bar of the four risk categories (bug classes), first used in "Bugs on the Horizon".
% Each argument is an overlay spec (use 0 for "never"):
%   N  = steps where the bar is neutral (all black text, light gray bars);
%   Sk = steps where risk k is on (purple text, thick purple bar); wins over N.
% On any other step, risk k is off (gray text, light gray bar).
% That's the 17 states: neutral, or any on/off combination of the 4 risks.
\newcommand{\riskTab}[4]{% #1 = neutral steps, #2 = on steps, #3 = heading, #4 = name
  {\color{sporkGrayttColor}\sporkSlideOnly<#1>{\usebeamercolor[fg]{normal text}}%
  \sporkSlideOnly<#2>{\color{sporkMainColor}}%
  \textbf{#3}\\#4\par}%
  \vspace{0.8mm}%
  {\color{sporkGrayttColor!35}\sporkSlideOnly<#2>{\color{sporkMainColor}}\rule{\linewidth}{1.2mm}}%
}
\newcommand{\riskBar}[5]{%
  \begin{columns}[T]
  \begin{column}{0.24\textwidth}\riskTab{#1}{#2}{Risk 1/4:}{Dataflow Logic}\end{column}
  \begin{column}{0.24\textwidth}\riskTab{#1}{#3}{Risk 2/4:}{Instruction Usage}\end{column}
  \begin{column}{0.24\textwidth}\riskTab{#1}{#4}{Risk 3/4:}{Inter-thread Sync}\end{column}
  \begin{column}{0.24\textwidth}\riskTab{#1}{#5}{Risk 4/4:}{Intra-thread Sync}\end{column}
  \end{columns}%
}

% fixed text={H}{D}: fixed text height/depth for a one-line node whose text is highlighted
% (\sporkGreen[only=<n>]{..} etc.) on some steps only. The highlight box is \fboxsep (1.5pt)
% taller and deeper than plain text, so without this the node and its text jitter between steps.

\tikzset{fixed text/.style 2 args={text height=#1, text depth=#2}}

% Trace event boxes (simplified InOutSets.tex look, black/white): a constructor box with small
% attached \sporkObIn (left) / \sporkObOut (right) boxes, named <name>-in / <name>-out.
% \trPre/\trPost: [#1] = extra Ob box options (e.g. spork green style), #2 = node name,
% #3 = main node options (placement), #4 = access line (e.g. read $\sporkx[\sporkz^*]$).
% \trFence: #1 = node name, #2 = main node options.
\tikzset{
  tr event/.style={rectangle, draw, thick, align=center, inner sep=1.5mm, minimum height=12mm},
  tr access/.style={tr event, minimum width=30mm},
  tr ob/.style={rectangle, draw, thick, inner sep=1mm, minimum height=7mm},
}
\newcommand{\trObGap}{0.8pt}  % gap between event and Ob boxes: one `thick` line
\newcommand{\trPre}[4][]{%
  \node (#2) [tr access, #3] {$\sporkTrPreAccess$\\#4};
  \node (#2-in) [tr ob, #1, anchor=east] at ([xshift=-\trObGap]#2.west) {$\sporkObIn$};}
\newcommand{\trPost}[4][]{%
  \node (#2) [tr access, #3] {$\sporkTrPostAccess$\\#4};
  \node (#2-out) [tr ob, #1, anchor=west] at ([xshift=\trObGap]#2.east) {$\sporkObOut$};}
\newcommand{\trFence}[2]{%
  \node (#1) [tr event, #2] {$\sporkTrFence$};
  \node (#1-in) [tr ob, anchor=east] at ([xshift=-\trObGap]#1.west) {$\sporkObIn$};
  \node (#1-out) [tr ob, anchor=west] at ([xshift=\trObGap]#1.east) {$\sporkObOut$};}

% (placement, color), #4 = event text below the constructor, then the set(s): \vtrPost #5 = Ob^out;
% \vtrFence #5 = Ob^in, #6 = Ob^out. Nodes <name>, <name>-in, <name>-out as \trPre etc.
