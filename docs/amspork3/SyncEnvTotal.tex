\magicSubsection{Lemma 1}{sec:SyncEnvTotal}

Define
\begin{align}
  & \sporktoEnv: \sporkTr^* \rightharpoonup \sporkEnv \notag \\
  & \sporktoEnv((\sporktr_1, ..., \sporktr_n)) = \varnothing \sporkConcatEnv \sporktr_1 \sporkConcatEnv ... \sporkConcatEnv \sporktr_n
  \label{eq:toEnv}
\end{align}
Define
\begin{align}
  & \sporkvVis : \sporkTr^* \times \sporkVNew \to \sporkEnvVis \notag \\
  & \sporkvVis(\sporktr^*, \sporkvNew) = \textbf{let } \sporkVert_d = \sporktrReach(\sporktr^*, \sporkvNew), \sporkQtlMask = \sporkvMask(\sporkvNew) \textbf{ in} \notag \\
  & \quad ( \bigcup_{(z, \sporktr_d) \in \sporkVert_d} \sporkObOut(\sporktr_d) \cap (\sporkTid \times \sporkQtlMask),
      \bigcup_{(z, \sporktr_d) \in \sporkVert_d} \sporkAwOut(\sporktr_d),
      \sporkQtlMask)
  \label{eq:vVis} \\
  & \sporktoEnvTotal: \sporkTr^* \to \sporkEnv \notag \\
  & \sporktoEnvTotal(\sporktr^*) = \{ (\sporkvEff(\sporkvNew), \sporkvVis(\sporktr^*, \sporkvNew)) \mid \sporkvNew \in \sporktrVNew(\sporktr^*, \sporkEnvEff) \}
  \label{eq:toEnvTotal}
\end{align}
Claim
\begin{align}
    \sporktr^* \in \textrm{dom}(\sporktoEnv) \implies \sporktoEnv(\sporktr^*) = \sporktoEnvTotal(\sporktr^*)
\end{align}
i.e. $\sporktoEnvTotal$ is a closed-form version of $\sporktoEnv$ with error checking omitted.
