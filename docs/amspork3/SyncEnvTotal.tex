\sporkSubsection{Closed-Form Environment}{sec:SyncEnvTotal}

Define
\begin{align}
  & \sporkvVis : \sporkTr^* \times \sporkVPost \to \sporkEnvVis \sporkComment{ ($\sporkTrPostAccess$ vertex to reachable outbound set union)} \notag \\
  & \sporkvVis(\sporktr^*, \sporkvPost) = (\varnothing, \varnothing, \sporkvMask(\sporkvPost)) \sporkConcatVis \sporktrReach(\sporktr^*, \sporkvPost)
  \label{eq:vVis} \\
  & \sporktoEnvTotal: \sporkTr^* \to \sporkEnv \sporkComment{ \sporkComment{ ($\sporkTrPostAccess$ become $\sporkenv$ elements; $\sporkEnvEff$ means no filter)}} \notag \\
  & \sporktoEnvTotal(\sporktr^*) = \{ (\sporkvEff(\sporkvPost), \sporkvVis(\sporktr^*, \sporkvPost)) \mid \sporkvPost \in \sporktrVPost(\sporktr^*, \sporkEnvEff) \}
  \label{eq:toEnvTotal}
\end{align}

\sporkMainKey{Step Lemma:}
For any $\sporktr^*$, $\sporktr_d$, and $\sporkvPost \in \sporktrVPost(\sporktr^*, \sporkEnvEff)$,
\begin{align}
  \text{(a)}\quad & \sporkvert_d \in \sporktrReach(\sporktr^*_\mathrm{cat}, \sporkvPost) \iff \sporksyncsWith(\sporkvVis(\sporktr^*, \sporkvPost), \sporktr_d) \\
  \text{(b)}\quad & \sporkvVis(\sporktr^*_\mathrm{cat}, \sporkvPost) = \sporkvVis(\sporktr^*, \sporkvPost) \sporkAugmentVis \sporktr_d
\end{align}
\emph{Proof.}
Let $\sporkQtlMask = \sporkvMask(\sporkvPost)$, $\sporkVert_r = \sporktrReach(\sporktr^*, \sporkvPost)$,
and $\sporkenvVis = \sporkvVis(\sporktr^*, \sporkvPost) = (\varnothing, \varnothing, \sporkQtlMask) \sporkConcatVis \sporkVert_r$.
\begin{itemize}
  \item[(S1)] $\sporktrReach(\sporktr^*_\mathrm{cat}, \sporkvPost) = \sporkVert_r \cup \{\sporkvert_d\}$
    if $\exists \sporkvert \in \sporkVert_r.\, \sporkEdgeArrow{\sporkvert}{\sporkvert_d} \in \sporktrE(\sporktr^*_\mathrm{cat}, \sporkQtlMask)$,
    and $\sporkVert_r$ otherwise.
    \sporkComment{(F3)}
  \item[(S2)] For $\sporkvert \in \sporktrV(\sporktr^*)$:
    $\sporkEdgeArrow{\sporkvert}{\sporkvert_d} \in \sporktrE(\sporktr^*_\mathrm{cat}, \sporkQtlMask) \iff
    \sporksyncsWith((\varnothing, \varnothing, \sporkQtlMask) \sporkConcatVis \{\sporkvert\}, \sporktr_d)$. \\
    \sporkComment{(unfold equations~\ref{eq:syncsWith}, \ref{eq:ConcatVis} and Section~\ref{sec:TraceEdges}; the second $\sporkTid \times \sporkQtlMask$ filter is redundant)}
  \item[(S3)] $(\exists \sporkvert \in \sporkVert_r.\, \sporksyncsWith((\varnothing, \varnothing, \sporkQtlMask) \sporkConcatVis \{\sporkvert\}, \sporktr_d))
    \iff \sporksyncsWith(\sporkenvVis, \sporktr_d)$. \\
    \sporkComment{($\sporksyncsWith$ is non-empty intersection tests; $\sporkConcatVis$ built $\sporkenvVis$ as unions of $\sporkob,\sporkaw$ in $\sporkvert\in\sporkVert_r$)}
\end{itemize}
(a) is S1--S3 chained. For (b), by S1 and (a),
\begin{align*}
  \sporkvVis(\sporktr^*_\mathrm{cat}, \sporkvPost) = \begin{cases}
    (\varnothing, \varnothing, \sporkQtlMask) \sporkConcatVis (\sporkVert_r \cup \{\sporkvert_d\}) = \sporkenvVis \sporkConcatVis \{\sporktr_d\} \;\sporkComment{(F1)}
      & \sporksyncsWith(\sporkenvVis, \sporktr_d) \\
    \sporkenvVis & \text{otherwise}
  \end{cases}
\end{align*}
which is $\sporkenvVis \sporkAugmentVis \sporktr_d$ (equation~\ref{eq:AugmentVis}). $\square$

\sporkMainKey{Closed-Form Lemma:}
If $\varnothing \sporkConcatEnv \sporktr^*$ is defined, then $\varnothing \sporkConcatEnv \sporktr^* = \sporktoEnvTotal(\sporktr^*)$;
i.e. $\sporktoEnvTotal$ is a closed form of $\sporkConcatEnv$ with error checking omitted.

\emph{Proof} by induction on $\abs{\sporktr^*}$.
Base: $\sporktoEnvTotal(()) = \varnothing$, as $\sporktrV(()) = \varnothing$.
Step: suppose $\varnothing \sporkConcatEnv \sporktr^*_\mathrm{cat}$ is defined.
Then so is the shorter fold, and by induction $\sporkenv = \varnothing \sporkConcatEnv \sporktr^* = \sporktoEnvTotal(\sporktr^*)$;
we show $\sporkenv \sporkConcatEnv \sporktr_d = \sporktoEnvTotal(\sporktr^*_\mathrm{cat})$.
Each old $\sporkvPost \in \sporktrVPost(\sporktr^*, \sporkEnvEff)$ keeps its $\sporkvEff$,
and by Step Lemma (b) its $\sporkvVis$ becomes $\sporkvVis(\sporktr^*, \sporkvPost) \sporkAugmentVis \sporktr_d$. By cases on $\sporktr_d$:
\begin{itemize}
  \item $\sporkTrFence$, $\sporkTrArrive$, $\sporkTrAwait$:
    no new $\sporkTrPostAccess$ vertex, so applying $\cdot \sporkAugmentVis \sporktr_d$ to the visibility record of every pair of $\sporktoEnvTotal(\sporktr^*)$
    gives $\sporktoEnvTotal(\sporktr^*_\mathrm{cat})$; this is equation~\ref{eq:envConcatSync}.
  \item $\sporkTrPreAccess$: no new $\sporkTrPostAccess$ vertex, and $\sporkAugmentVis \sporktr_d$ is the identity (F2),
    so $\sporktoEnvTotal(\sporktr^*_\mathrm{cat}) = \sporkenv$, matching equation~\ref{eq:envConcatPre} (defined by assumption).
  \item $\sporkTrPostAccess$: old pairs are unchanged (F2).
    The new $\sporkvPost = \sporkvert_d$ reaches only itself (0-length path; F3),
    so it contributes $(\sporkvEff(\sporktr_d), (\varnothing, \varnothing, \sporkvMask(\sporktr_d)) \sporkConcatVis \{\sporktr_d\})$:
    the pair added by equation~\ref{eq:envConcatPost}. $\square$
\end{itemize}
