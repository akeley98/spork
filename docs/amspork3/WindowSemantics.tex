\textrm{Wrange}(z) = \{z\} \quad

  \textrm{rhs} = \sporkWunroll(\sporkwz^*)
}
{
  \sporkWunroll(\sporkwz, \sporkwz^*) = \textrm{lhs} \cup \textrm{rhs}
